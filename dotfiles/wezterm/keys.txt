# WezTerm key sheet. ~/.zshrc prints it in each new tab; run `wezkeys` to see
# it again. Each [Group] starts a section, and each "action = keys" line is one
# binding. Keep it in step with config.keys in wezterm.lua.

[Tabs]
new tab = Ctrl+Shift+T
close tab = Ctrl+Shift+W
rename tab = Ctrl+Shift+E
PowerShell/cmd = Ctrl+Shift+O
go to tab 1-8 = Ctrl+Shift+1..8
last tab = Ctrl+Shift+9
next tab = Ctrl+Tab
previous tab = Ctrl+Shift+Tab
move tab left = Ctrl+Shift+PgUp
move tab right = Ctrl+Shift+PgDn

[Panes]
split right = Ctrl+Shift+Alt+5
split down = Ctrl+Shift+Alt+'
focus pane = Ctrl+Shift+Arrow
resize pane = Ctrl+Shift+Alt+Arrow
zoom pane = Ctrl+Shift+Z

[Text]
copy = Ctrl+Shift+C
paste = Ctrl+Shift+V
search = Ctrl+Shift+F
copy mode = Ctrl+Shift+X
quick select = Ctrl+Shift+Space
clear scrollback = Ctrl+Shift+K
scroll up = Shift+PgUp
scroll down = Shift+PgDn

[Window]
new window = Ctrl+Shift+N
full screen = Alt+Enter
bigger font = Ctrl+=
smaller font = Ctrl+-
reset font = Ctrl+0
command palette = Ctrl+Shift+P
