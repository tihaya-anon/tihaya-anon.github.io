This is the Bash setup I use day to day, rather than a raw backup. It keeps shell startup responsibilities in .bashrc, short command definitions in .bash_aliases, machine-local environment in small private files, and prompt presentation in Starship.

The files load in this order:

  1. .bashrc returns immediately for non-interactive shells, configures history and completion, then loads optional tools when they are installed.
  2. .bash_aliases adds command shortcuts once. It is sourced after the base ls and grep aliases, so there are no competing definitions.
  3. .bash_env and .mirrors provide private exports and host-specific mirror settings without putting secrets in the public config.
  4. Starship replaces the default Bash prompt when starship is available.

The configuration targets Debian or Ubuntu, but missing optional tools such as ble.sh, Homebrew, NVM, SDKMAN, Cargo, Zoxide, and Starship are safely skipped. Copy the sections you need, and adjust the HOME-relative paths for your installation.

Private environment variables
#

Keep machine-local secrets and account-specific exports in ~/.bash_env, then load it only for interactive shells. This keeps values such as OPENAPI_KEY out of .bashrc and out of this repository. The file can contain ordinary shell exports, for example export OPENAPI_KEY="..."; it should never be committed or pasted into a public post. Restricting it with chmod 600 ~/.bash_env is appropriate when it contains credentials.

.bashrc
#

  1# ~/.bashrc: executed by bash(1) for non-login shells.
  2
  3# Only configure interactive shells.
  4case $- in
  5    *i*) ;;
  6    *) return ;;
  7esac
  8
  9# Ignore duplicate and space-prefixed history entries, and retain it across shells.
 10HISTCONTROL=ignoreboth
 11shopt -s histappend
 12HISTSIZE=1000
 13HISTFILESIZE=2000
 14shopt -s checkwinsize
 15
 16path_prepend() {
 17    [ -d "$1" ] || return
 18    case ":$PATH:" in
 19        *":$1:"*) ;;
 20        *) PATH="$1:$PATH" ;;
 21    esac
 22}
 23
 24path_append() {
 25    [ -d "$1" ] || return
 26    case ":$PATH:" in
 27        *":$1:"*) ;;
 28        *) PATH="$PATH:$1" ;;
 29    esac
 30}
 31
 32# Make `less` more useful for non-text files when lesspipe is installed.
 33[ -x /usr/bin/lesspipe ] && eval "$(SHELL=/bin/sh lesspipe)"
 34
 35# Used by the fallback prompt on Debian systems.
 36if [ -z "${debian_chroot:-}" ] && [ -r /etc/debian_chroot ]; then
 37    debian_chroot=$(< /etc/debian_chroot)
 38fi
 39
 40# Use a colored fallback prompt when the terminal supports it.
 41case "$TERM" in
 42    xterm-color|*-256color) color_prompt=yes ;;
 43esac
 44
 45if [ -n "$force_color_prompt" ]; then
 46    if [ -x /usr/bin/tput ] && tput setaf 1 >&/dev/null; then
 47        color_prompt=yes
 48    else
 49        color_prompt=
 50    fi
 51fi
 52
 53if [ "$color_prompt" = yes ]; then
 54    PS1='${debian_chroot:+($debian_chroot)}\[\033[01;32m\]\u@\h\[\033[00m\]:\[\033[01;34m\]\w\[\033[00m\]\$ '
 55else
 56    PS1='${debian_chroot:+($debian_chroot)}\u@\h:\w\$ '
 57fi
 58unset color_prompt force_color_prompt
 59
 60# Set the terminal title in xterm-compatible terminals.
 61case "$TERM" in
 62xterm*|rxvt*)
 63    PS1="\[\e]0;${debian_chroot:+($debian_chroot)}\u@\h: \w\a\]$PS1"
 64    ;;
 65esac
 66
 67# Base aliases are defined here once; personal shortcuts stay in .bash_aliases.
 68if [ -x /usr/bin/dircolors ]; then
 69    test -r ~/.dircolors && eval "$(dircolors -b ~/.dircolors)" || eval "$(dircolors -b)"
 70    alias ls='ls --color=auto'
 71    alias grep='grep --color=auto'
 72    alias fgrep='fgrep --color=auto'
 73    alias egrep='egrep --color=auto'
 74fi
 75
 76# Listing views are also base shell behavior.
 77alias ll='ls -alF'
 78alias la='ls -A'
 79alias l='ls -CF'
 80
 81# Send a desktop notification after a long-running command.
 82alias alert='notify-send --urgency=low -i "$([ $? = 0 ] && echo terminal || echo error)" "$(history|tail -n1|sed -e '\''s/^\s*[0-9]\+\s*//;s/[;&|]\s*alert$//'\'')"'
 83
 84# Personal aliases and shell functions.
 85if [ -f ~/.bash_aliases ]; then
 86    . ~/.bash_aliases
 87fi
 88
 89# Private machine-local exports and mirror settings.
 90[ -r "$HOME/.bash_env" ] && source "$HOME/.bash_env"
 91[ -r "$HOME/.mirrors" ] && source "$HOME/.mirrors"
 92
 93# Programmable completion, if the distribution has installed it.
 94if ! shopt -oq posix; then
 95    if [ -f /usr/share/bash-completion/bash_completion ]; then
 96        . /usr/share/bash-completion/bash_completion
 97    elif [ -f /etc/bash_completion ]; then
 98        . /etc/bash_completion
 99    fi
100fi
101
102# Optional tools. Their guards keep shell startup quiet on a new machine.
103[ -r "$HOME/.local/share/blesh/ble.sh" ] && source "$HOME/.local/share/blesh/ble.sh"
104[ -r "$HOME/.local/bin/env" ] && source "$HOME/.local/bin/env"
105
106export UV_CACHE_DIR="${UV_CACHE_DIR:-/tmp/.cache/uv}"
107
108path_prepend "$HOME/.local/bin"
109path_append "/opt/nvim-linux-x86_64/bin"
110path_prepend "$HOME/.terragrunt/bin"
111path_append "$HOME/.pulumi/bin"
112
113[ -r "$HOME/.cargo/env" ] && source "$HOME/.cargo/env"
114
115export NVM_DIR="$HOME/.nvm"
116[ -s "$NVM_DIR/nvm.sh" ] && . "$NVM_DIR/nvm.sh"
117[ -s "$NVM_DIR/bash_completion" ] && . "$NVM_DIR/bash_completion"
118
119export SDKMAN_DIR="$HOME/.sdkman"
120[ -s "$SDKMAN_DIR/bin/sdkman-init.sh" ] && source "$SDKMAN_DIR/bin/sdkman-init.sh"
121
122# Homebrew adjusts PATH and related variables.
123if [ -x /home/linuxbrew/.linuxbrew/bin/brew ]; then
124    eval "$(/home/linuxbrew/.linuxbrew/bin/brew shellenv bash)"
125fi
126export HOMEBREW_NO_ENV_HINTS=1
127
128# Zoxide replaces repeated `cd` navigation, when installed.
129command -v zoxide >/dev/null 2>&1 && eval "$(zoxide init bash)"
130
131# Starship owns the final prompt, when installed.
132command -v starship >/dev/null 2>&1 && eval "$(starship init bash)"
133
134unset -f path_prepend path_append

These are intentionally additive: ls, ll, and the grep family are already owned by .bashrc. ping is capped at five packets to avoid accidentally leaving it running; use command ping for the system default behavior.

33gr() {
34    if [ "$#" -eq 0 ]; then
35        git remote -v
36    else
37        git remote "$@"
38    fi
39}
40alias gcp='git cherry-pick'
41alias gst='git stash'
42# Show the five most recent commits unless a count is supplied.
43glg() {
44    if [ "$#" -eq 0 ]; then
45        git log --oneline -n 5
46    else
47        git log --oneline -n "$@"
48    fi
49}
50# Create merge commits without opening an editor, except when explicitly requested.
51gmg() {
52    local arg
53
54    for arg in "$@"; do
55        if [ "$arg" = '-e' ]; then
56            git merge "$@"
57            return
58        fi
59    done
60
61    git merge --no-edit "$@"
62}
